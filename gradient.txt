CELL 3
def funtion(x) : 
    return (x+3)**2 

def derivative(x) : 
    return 2*(x+3)

CELL 4
x = 2 

learing_rate = 0.1 

iterations = 50 

x_values = [] 
y_values = [] 


for i in range(iterations) : 

    x_values.append(x)
    y_values.append(funtion(x)) 


    gradient = derivative(x) 


    x = x - learing_rate*gradient 


    print("Iteration : " , i+1 ,  "x :" , x , "y :" , funtion(x)) 




print("Local Minima :") 

print("x = " , x)

print("y = " , funtion(x))

x_range = np.linspace(-5, 4, 100)

plt.xlabel('x')
plt.ylabel('y')
plt.title('Gradient Descent')
plt.legend()
plt.grid()
